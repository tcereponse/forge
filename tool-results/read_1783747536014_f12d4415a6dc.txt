   607→const markdownApp = `import { useState } from 'react'
   608→
   609→const renderMd = (md: string): string => {
   610→  let html = md
   611→    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
   612→    .replace(/^### (.+)$/gm, '<h3 class="text-lg font-semibold text-cyan-300 mt-3 mb-1">$1</h3>')
   613→    .replace(/^## (.+)$/gm, '<h2 class="text-xl font-bold text-cyan-400 mt-3 mb-1">$1</h2>')
   614→    .replace(/^# (.+)$/gm, '<h1 class="text-2xl font-bold text-cyan-400 mt-3 mb-1">$1</h1>')
   615→    .replace(/\\*\\*(.+?)\\*\\*/g, '<strong>$1</strong>')
   616→    .replace(/\\*(.+?)\\*/g, '<em>$1</em>')
   617→    .replace(/\`(.+?)\`/g, '<code class="bg-slate-800 px-1 rounded text-cyan-300">$1</code>')
   618→    .replace(/^- (.+)$/gm, '<li class="ml-4 list-disc">$1</li>')
   619→    .replace(/\\n/g, '<br/>')
   620→  return html
   621→}

   623→interface Note {
   624→  id: number
   625→  title: string
   626→  body: string
   627→}
   628→
   629→export default function App() {

   638→    const next = { ...active, body, title }
   639→    setActive(next)
   640→    setNotes(notes.map(n => (n.id === next.id ? next : n)))
   641→  }
   642→
   643→  const create = () => {
   644→    const n = { id: Date.now(), title: 'Nouvelle note', body: '# Nouvelle note' }
   645→    setNotes([...notes, n]); setActive(n)
   646→  }
   647→
   648→  return (
   649→    <div className="min-h-screen bg-slate-950 text-slate-100 p-4 max-w-md mx-auto">
   650→      <div className="flex items-center justify-between mb-4">
   651→        <h1 className="text-2xl font-bold text-cyan-400">MarkdownNotes</h1>

   653→          + Nouvelle
   654→        </button>
   655→      </div>
   656→
   657→      <div className="flex gap-2 mb-3 overflow-x-auto pb-2">

   660→            key={n.id}
   661→            onClick={() => setActive(n)}
   662→            className={\`px-3 py-1.5 rounded-lg text-xs whitespace-nowrap \${
   663→              active?.id === n.id ? 'bg-cyan-500 text-slate-950' : 'bg-slate-800 text-slate-400'
   664→            }\`}
   665→          >
   666→            {n.title}

   669→      </div>

   671→      {active && (

   675→            onChange={e => update(e.target.value)}

   689→// ============ 8. QuizMaster ============
   690→const quizApp = `import { useEffect, useState } from 'react'
   691→
   692→interface Q {
   693→  q: string
   694→  opts: string[]
   695→  ans: number
   696→}
   697→
   698→const QUESTIONS: Q[] = [
   699→  { q: 'Capitale du Japon ?', opts: ['Séoul', 'Tokyo', 'Pékin', 'Bangkok'], ans: 1 },
   700→  { q: '2 + 2 × 3 = ?', opts: ['12', '8', '10', '6'], ans: 1 },
   701→  { q: 'Langage de React ?', opts: ['Python', 'Go', 'JavaScript', 'Rust'], ans: 2 },
   702→  { q: 'Couleur du ciel le jour ?', opts: ['Rouge', 'Bleu', 'Vert', 'Noir'], ans: 1 },
   703→  { q: 'Plus grand océan ?', opts: ['Atlantique', 'Indien', 'Pacifique', 'Arctique'], ans: 2 },
   704→]
   705→
   706→export default function App() {
   707→  const [idx, setIdx] = useState(0)
   708→  const [score, setScore] = useState(0)
   709→  const [picked, setPicked] = useState<number | null>(null)
   710→  const [time, setTime] = useState(15)
   711→  const [done, setDone] = useState(false)
   712→
   713→  useEffect(() => {
   714→    if (done || picked !== null) return
   715→    if (time <= 0) { setPicked(-1); return }
   716→    const t = setTimeout(() => setTime(time - 1), 1000)
   717→    return () => clearTimeout(t)
   718→  }, [time, picked, done])
   719→
   720→  const choose = (i: number) => {
   721→    if (picked !== null) return
   722→    setPicked(i)
   723→    if (i === QUESTIONS[idx].ans) setScore(s => s + 1)
   724→  }
   725→
   726→  const next = () => {
   727→    if (idx + 1 >= QUESTIONS.length) { setDone(true); return }
   728→    setIdx(idx + 1); setPicked(null); setTime(15)
   729→  }
   730→
   731→  const restart = () => { setIdx(0); setScore(0); setPicked(null); setTime(15); setDone(false) }
   732→
   733→  if (done) {
   734→    return (
   735→      <div className="min-h-screen bg-slate-950 text-slate-100 p-4 flex flex-col items-center justify-center">
   736→        <h1 className="text-3xl font-bold text-cyan-400 mb-2">Terminé !</h1>
   737→        <p className="text-6xl font-bold my-4">{score} / {QUESTIONS.length}</p>
   738→        <p className="text-slate-400 mb-6">
   739→          {score === QUESTIONS.length ? '🏆 Parfait !' : score >= 3 ? '👍 Bien joué' : '💪 Continue'}
   740→        </p>
   741→        <button onClick={restart} className="bg-cyan-500 text-slate-950 px-6 py-3 rounded-xl font-semibold">
   742→          Rejouer
   743→        </button>
   744→      </div>
   745→    )
   746→  }
   747→
   748→  const cur = QUESTIONS[idx]
   749→  return (
   750→    <div className="min-h-screen bg-slate-950 text-slate-100 p-4 max-w-md mx-auto flex flex-col">
   751→      <div className="flex justify-between items-center mb-4">
   752→        <span className="text-sm text-slate-400">Question {idx + 1}/{QUESTIONS.length}</span>
   753→        <span className={\`text-sm font-mono px-2 py-1 rounded \${
   754→          time <= 5 ? 'bg-red-500/20 text-red-400' : 'bg-slate-800 text-cyan-400'
   755→        }\`}>{time}s</span>
   756→      </div>
   757→
   758→      <div className="h-1 bg-slate-800 rounded-full mb-6">
   759→        <div className="h-full bg-cyan-500 rounded-full transition-all"
   760→          style={{ width: \`\${((idx) / QUESTIONS.length) * 100}%\` }} />
   761→      </div>
   762→
   763→      <h2 className="text-xl font-semibold mb-6">{cur.q}</h2>
   764→
   765→      <div className="space-y-3">
   766→        {cur.opts.map((o, i) => {
   767→          let cls = 'bg-slate-900 border-slate-800'
   768→          if (picked !== null) {
   769→            if (i === cur.ans) cls = 'bg-cyan-500/20 border-cyan-500'
   770→            else if (i === picked) cls = 'bg-red-500/20 border-red-500'
   771→          }
   772→          return (
   773→            <button
   774→              key={i}
   775→              onClick={() => choose(i)}
   776→              disabled={picked !== null}
   777→              className={\`w-full text-left p-4 rounded-xl border \${cls} transition-colors\${
   778→                picked === null ? ' active:scale-98' : ''
   779→              }\`}
   780→            >
   781→              <span className="text-cyan-400 mr-2">{String.fromCharCode(65 + i)}.</span>
   782→              {o}
   783→            </button>
   784→          )
   785→        })}
   786→      </div>
   787→
   788→      {picked !== null && (
   789→        <button onClick={next} className="mt-6 bg-cyan-500 text-slate-950 py-3 rounded-xl font-semibold">
   790→          {idx + 1 >= QUESTIONS.length ? 'Voir le résultat' : 'Question suivante'}
   791→        </button>
   792→      )}
   793→    </div>
   794→  )
   795→}`
