    22→class KirovLogger {
    23→    static levels = { DEBUG: 0, INFO: 1, WARN: 2, ERROR: 3 };
    24→    static currentLevel = CONFIG.DEBUG_MODE ? 0 : 1;
    25→    static metrics = { promptsInjected: 0, captures: 0, bridgeErrors: 0 };

    27→    static _log(levelName, ...args) {
    28→        if (this.levels[levelName] >= this.currentLevel) {
    29→            const prefix = `[KIROV3] [${levelName}]`;
    30→            if (levelName === 'ERROR') console.error(prefix, ...args);
    31→            else if (levelName === 'WARN') console.warn(prefix, ...args);
    32→            else console.log(prefix, ...args);
    33→        }
    34→    }
    35→    static debug(...args) { this._log('DEBUG', ...args); }
    36→    static info(...args) { this._log('INFO', ...args); }
    37→    static warn(...args) { this._log('WARN', ...args); }
    38→    static error(...args) { this._log('ERROR', ...args); }
    39→    
    40→    static trackMetric(name) {
    41→        if (this.metrics[name] !== undefined) this.metrics[name]++;
    42→    }

    46→// 3. UTILITAIRES (Crypto, Validation)
    47→// ==========================================
    48→const Utils = {
    49→    async sha256(text) {
    50→        if (!text) return "";
    51→        const msgBuffer = new TextEncoder().encode(text);
    52→        const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
    53→        const hashArray = Array.from(new Uint8Array(hashBuffer));
    54→        return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    55→    },
    56→    validateBridgeResponse(data) {
    57→        if (!data || typeof data !== 'object') return false;
    58→        if (!('status' in data)) return false; // Validation de schéma minimaliste
    59→        return true;
    60→    }
    61→};

    63→// ==========================================
    64→// 4. EVENT BUS (Pattern Observer)
    65→// ==========================================
    66→class EventBus {
    67→    constructor() { this.listeners = {}; }
    68→    on(event, callback) {
    69→        if (!this.listeners[event]) this.listeners[event] = [];
    70→        this.listeners[event].push(callback);
    71→    }
    72→    emit(event, data) {
    73→        if (this.listeners[event]) {
    74→            this.listeners[event].forEach(cb => {
    75→                try { cb(data); } catch(e) { KirovLogger.error(`EventBus Error on ${event}:`, e); }
    76→            });
    77→        }
    78→    }
    79→}

   120→// 6. BRIDGE CLIENT (Resilience Network)
   121→// ==========================================
   122→class BridgeClient {
   123→    constructor() {
   124→        this.consecutiveFailures = 0;
   125→        this.circuitOpen = false;
   126→        this.currentAbortController = null;
   127→        this.pollingIntervalId = null;
   128→        this.currentInterval = CONFIG.POLLING_BASE_INTERVAL;
   129→    }
   130→
   131→    startPolling() {
   132→        if (this.pollingIntervalId) clearTimeout(this.pollingIntervalId);
   133→        this._pollCycle();
   134→    }
   135→
   136→    async _pollCycle() {
   137→        if (!this.circuitOpen) await this.poll();
   138→        
   139→        // Adaptive polling : ralentit si hors-ligne pour économiser CPU/Réseau
   140→        this.currentInterval = state.get('isOffline') ? 10000 : CONFIG.POLLING_BASE_INTERVAL;
   141→        this.pollingIntervalId = setTimeout(() => this._pollCycle(), this.currentInterval);
   142→    }
   143→
   144→    async poll() {
   145→        // Annulation de la requête précédente si elle pend (Conflits Promises)
   146→        if (this.currentAbortController) this.currentAbortController.abort();
   147→        this.currentAbortController = new AbortController();
   148→
   149→        try {
   150→            const platformName = platform.config.host || 'fallback';
   151→            const res = await fetch(`${CONFIG.SERVER_URL}/v1/bridge/poll?platform=${platformName}`, {
   152→                signal: this.currentAbortController.signal
   153→            });
   154→            
   155→            if (!res.ok) throw new Error(`HTTP ${res.status}`);
   156→            const data = await res.json();
   157→            
   158→            if (!Utils.validateBridgeResponse(data)) throw new Error("Invalid schema");
   159→
   160→            this._handleSuccess();
   161→            bus.emit('BRIDGE_SYNC', data);
   162→            
   163→            if (data.phase_num === 4) this.pollLogs();
   164→
   165→        } catch (e) {
   166→            if (e.name !== 'AbortError') this._handleFailure(e);
   167→        }
   168→    }
   169→
   170→    async sendCapture(payload) {
   171→        const strPayload = JSON.stringify(payload);
   172→        if (strPayload.length > CONFIG.MAX_PAYLOAD_SIZE) {
   173→            KirovLogger.warn("Payload over 500KB, truncating or dropping.");
   174→            return;
   175→        }
   176→
   177→        if (state.get('isOffline')) {
   178→            KirovLogger.info("Offline: Queueing payload.");
   179→            const q = state.get('queue');
   180→            q.push(payload);
   181→            state.set('queue', q);
   182→            return;
   183→        }
   184→
   185→        try {
   186→            const res = await fetch(`${CONFIG.SERVER_URL}/v1/bridge/callback`, {
   187→                method: 'POST',
   188→                headers: { 'Content-Type': 'application/json' },
   189→                body: strPayload
   190→            });
   191→            if (res.ok) {
   192→                KirovLogger.trackMetric('captures');
   193→                bus.emit('CAPTURE_SENT', payload);
   194→            }
   195→        } catch (e) {
   196→            KirovLogger.error("Failed to send capture, queueing.", e);
   197→            const q = state.get('queue');
   198→            q.push(payload);
   199→            state.set('queue', q);
   200→            this._handleFailure(e);
   201→        }
   202→    }
   203→
   204→    async flushQueue() {
   205→        const q = state.get('queue');
   206→        if (q.length === 0) return;
   207→        KirovLogger.info(`Flushing queue: ${q.length} items`);
   208→        
   209→        const items = [...q];
   210→        state.set('queue', []);
   211→        for (const item of items) await this.sendCapture(item);
   212→    }
   213→
   214→    async pollLogs() {
   215→        try {
   216→            const res = await fetch(`${CONFIG.SERVER_URL}/v1/logs`);
   217→            if (!res.ok) return;
   218→            const data = await res.json();
   219→            bus.emit('LOGS_SYNC', data.logs || []);
   220→        } catch (e) { KirovLogger.warn("Log polling failed", e); }
   221→    }
   222→
   223→    _handleSuccess() {
   224→        this.consecutiveFailures = 0;
   225→        if (state.get('isOffline')) {
   226→            KirovLogger.info("Bridge reconnected.");
   227→            state.set('isOffline', false);
   228→            bus.emit('BRIDGE_ONLINE');
   229→            this.flushQueue();
   230→        }
   231→    }
   232→
   233→    _handleFailure(err) {
   234→        this.consecutiveFailures++;
   235→        KirovLogger.trackMetric('bridgeErrors');
   236→        KirovLogger.warn(`Bridge error (${this.consecutiveFailures}):`, err);
   237→
   238→        // Circuit Breaker : Stop au bout de 5 échecs
   239→        if (this.consecutiveFailures >= CONFIG.MAX_RETRIES && !state.get('isOffline')) {
   240→            KirovLogger.error("Circuit breaker triggered! Bridge is OFFLINE.");
   241→            state.set('isOffline', true);
   242→            bus.emit('BRIDGE_OFFLINE');
   243→        }
   244→    }
   245→}

   626→        const forbiddenColors = ['purple-', 'indigo-', 'violet-'];

   632→        const violations = [];
   633→        if (content.includes('.jsx')) violations.push('Extension .jsx détectée au lieu de .tsx (Règle 8).');
   634→        // Détection structure sous-dossier parasite
   635→        if (content.match(/client\/index\.html|frontend\/index\.html|web\/index\.html/)) {
   636→            violations.push('⚠️ STRUCTURE INVALIDE : index.html dans un sous-dossier (client/frontend/web). Doit être à la racine ! (Règle 11)');
   637→        }
   638→        // Détection PostCSS CJS dans ESM
   639→        if (content.includes('module.exports =') && content.includes('"type": "module"')) {
   640→            violations.push('⚠️ POSTCSS ESM CONFLICT : module.exports détecté dans un projet ESM. Utilise export default (Règle 12).');

   644→            violations.push('⚠️ VITE CONFIG : Le mot "import" est manquant en début de ligne (Règle 13).');
   645→        }
   646→        // NOTE : Détection pnpm désactivée car pnpm est le gestionnaire de dépendances standard et approuvé de la Forge globale !
   647→        return violations;
   648→    }

   658→    startObserving() {
   659→        const container = platform.getElement([platform.config.messageContainer, 'body']);
   660→        if (!container) return setTimeout(() => this.startObserving(), 1000);
   661→
   662→        KirovLogger.info("Starting Output Scanner Observer.");
   663→        this.observer = new MutationObserver(() => {
   664→            if (this.timeout) clearTimeout(this.timeout);
   665→            // Debounce de 500ms pour ne pas saturer le CPU pendant que l'IA tape
   666→            this.timeout = setTimeout(() => this.parseLatestMessage(), 500);
   667→        });
   668→        this.observer.observe(container, { childList: true, characterData: true, subtree: true });
   669→    }

   671→    isStreamingComplete(msgElement) {
   672→        // Multi-critères pour définir si l'IA a fini de générer
   673→        const hasSpinner = !!document.querySelector('.ds-spinner, .generating, button[aria-label*="Stop" i]');
   674→        return !hasSpinner;
   675→    }

   677→    async parseLatestMessage() {
   678→        const messages = platform.getElements(platform.config.messageBlock);
   679→        if (messages.length === 0) return;
   680→        
   681→        const lastMsg = messages[messages.length - 1];
   682→        if (!this.isStreamingComplete(lastMsg)) return;
   683→
   684→        // Clone du message pour le nettoyer sans modifier le DOM visible
   685→        const clone = lastMsg.cloneNode(true);
   686→        // Supprimer les boutons de l'interface qui génèrent "Copier", "Télécharger"
   687→        const uiElements = clone.querySelectorAll('button, svg, [role="button"], [class*="action"], [class*="copy"]');
   688→        uiElements.forEach(el => el.remove());
   689→
   690→        // FIX SOUVERAIN : Préserver les sauts de ligne car textContent les écrase sur les noeuds détachés
   691→        clone.querySelectorAll('p, div, li').forEach(el => el.appendChild(document.createTextNode('\n')));
   692→        clone.querySelectorAll('br').forEach(el => el.replaceWith('\n'));
   693→
   694→        let content = clone.innerText || clone.textContent || "";
   695→        // Filtrage Regex agressif pour les restes de l'UI DeepSeek (ignorer la casse, sur chaque ligne)
   696→        content = content.replace(/^(text|copier|télécharger|copy|download|exécuter)$/gmi, '').trim();
   697→
   698→        if (content.length < 50) return;
   699→
   700→        const contentHash = await Utils.sha256(content);
   701→        if (contentHash === state.get('lastOutputHash')) return; // Déduplication
   702→
   703→        if (content.match(/Fichier\s*:/i) || content.includes("FILE:") || content.includes("\`\`\`") || content.includes("PRD")) {
   704→            
   705→            // VALIDATION G50+
   706→            const phase = state.get('currentPhase');
   707→            const validationResult = { isValid: true };
   708→            
   709→            // On désactive la validation du Scaffold pour P5 et P6 (Refactoring Chirurgical)
   710→            if (phase >= 2 && phase < 5) {
   711→                const missing = ValidationOrchestrator.validateScaffold(content);
   712→                const designViolations = ValidationOrchestrator.validateDesign(content);
   713→                const importViolations = ValidationOrchestrator.validateImports(content);
   714→                
   715→                if (missing.length > 0) {
   716→                    KirovLogger.warn(`Scaffold incomplet: ${missing.join(', ')}`);
   717→                    bus.emit('UI_TOAST', `⚠️ Scaffold incomplet: ${missing.length} fichiers absents`);
   718→                    validationResult.isValid = false;
   719→                }
   720→                if (designViolations.length > 0) {
   721→                    KirovLogger.warn(`Violation Design: ${designViolations.join(', ')}`);
   722→                    bus.emit('UI_TOAST', `🎨 Violation Design: Couleur interdite détectée`);
   723→                }
   724→                if (importViolations.length > 0) {
   725→                    KirovLogger.warn(`Violation Imports: ${importViolations.join(', ')}`);
   726→                    bus.emit('UI_TOAST', `⚠️ Violation Import: Extension invalide (.jsx)`);
   727→                }
   728→            }
   729→
   730→            state.set('lastOutputHash', contentHash);
   731→            KirovLogger.info("New valid code block captured.");
   732→            bridge.sendCapture({ 
   733→                content: content,
   734→                validation: validationResult
   735→            });
   736→        }
   737→    }

   818→
