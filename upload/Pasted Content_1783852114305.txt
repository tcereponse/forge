def log(msg):

# --- Extraction automatique de app_ui.zip si present ---
try:
    ui_zip_path = os.path.join(BASE_DIR, "app_ui.zip")
    out_dir = os.path.join(BASE_DIR, "out")
    if os.path.exists(ui_zip_path):
        log(f"📦 Extraction automatique de {ui_zip_path} vers {out_dir}...")
        # Vider d'abord le dossier out pour eviter les melanges
        if os.path.exists(out_dir):
            shutil.rmtree(out_dir, ignore_errors=True)
        os.makedirs(out_dir, exist_ok=True)
        with zipfile.ZipFile(ui_zip_path, 'r') as zip_ref:
            zip_ref.extractall(out_dir)
        log("✅ Extraction UI terminee avec succes.")
except Exception as e:
    log(f"⚠️ Erreur lors de l'extraction de app_ui.zip: {e}")

@app.route('/api/projects', methods=['GET'])
def list_projects():
    conn = get_db()
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM Project ORDER BY createdAt DESC")
    rows = cursor.fetchall()
    projects = []
    for r in rows:
        p = dict(r)
        try:
            p["features"] = json.loads(p.get("features", "[]"))
        except:
            p["features"] = []
        try:
            p["selectedPacks"] = json.loads(p.get("selectedPacks", "[]"))
        except:
            p["selectedPacks"] = []
        projects.append(p)
    conn.close()
    return jsonify({"success": True, "projects": projects})

# --- COMPILATION DE SUTURE D'APK SOUVERAIN (SANS JDK NI SDK ANDROID) ---

def _rebuild_apk_with_patches(shell_apk, output_apk, assets_dir, authority):
    shell_entries = {}
    with zipfile.ZipFile(shell_apk, 'r') as zin:
        for item in zin.infolist():
            shell_entries[item.filename] = zin.read(item.filename)

    # file_paths.xml d'autorisation locale
    file_paths_xml = """<?xml version="1.0" encoding="utf-8"?>
<paths xmlns:android="http://schemas.android.com/apk/res/android">
    <external-path name="external_files" path="." />
    <files-path name="internal_files" path="." />
    <cache-path name="cache_files" path="." />
    <external-cache-path name="external_cache" path="." />
    <external-files-path name="external_app_files" path="." />
</paths>"""
    shell_entries['res/xml/file_paths.xml'] = file_paths_xml.encode('utf-8')

    # Remplacement de l'autorité FileProvider dans AndroidManifest.xml binaire
    if 'AndroidManifest.xml' in shell_entries:
        manifest_bytes = shell_entries['AndroidManifest.xml']
        new_auth_bytes = authority.encode('utf-16-le')
        OLD_AUTHORITIES = [
            b"com.elite.bestmode.provider",
            b"com.elite.app.provider",
            b"com.elite.provider",
        ]
        for old_auth in OLD_AUTHORITIES:
            old_auth_utf16 = old_auth.decode('ascii').encode('utf-16-le')
            if old_auth_utf16 in manifest_bytes:
                if len(new_auth_bytes) <= len(old_auth_utf16):
                    padded = new_auth_bytes + b'\x00\x00' * ((len(old_auth_utf16) - len(new_auth_bytes)) // 2)
                    manifest_bytes = manifest_bytes.replace(old_auth_utf16, padded, 1)
                else:
                    manifest_bytes = manifest_bytes.replace(old_auth_utf16, new_auth_bytes[:len(old_auth_utf16)], 1)
                break
        shell_entries['AndroidManifest.xml'] = manifest_bytes

    # Injecter les nouveaux assets Web du projet (dist/)
    for root, _, files in os.walk(assets_dir):
        for fname in sorted(files):
            full_path = os.path.join(root, fname)
            rel = os.path.relpath(full_path, assets_dir).replace('\\', '/')
            arc = f'assets/www/{rel}'
            with open(full_path, 'rb') as fp:
                shell_entries[arc] = fp.read()

    # Réassemblage
    out_buf = tempfile.SpooledTemporaryFile(max_size=100 * 1024 * 1024)
    with zipfile.ZipFile(out_buf, 'w') as zout:
        sorted_entries = sorted(shell_entries.items(), key=lambda x: 0 if x[0] == 'resources.arsc' else 1)
        for arc_name, data in sorted_entries:
            zinfo = zipfile.ZipInfo(arc_name)
            zinfo.compress_type = zipfile.ZIP_STORED if (
                arc_name == 'AndroidManifest.xml' or 
                arc_name == 'resources.arsc' or 
                arc_name.endswith('.png') or 
                arc_name.endswith('.jpg')
            ) else zipfile.ZIP_DEFLATED
            
            # Zipalign 4 octets
            if zinfo.compress_type == zipfile.ZIP_STORED:
                offset = out_buf.tell()
                header_size = 30 + len(zinfo.filename.encode('utf-8')) + len(zinfo.extra)
                padding = (4 - ((offset + header_size) % 4)) % 4
                if padding > 0:
                    padding += 4
                    zinfo.extra += b'\x35\xd9' + (padding - 4).to_bytes(2, 'little') + (b'\x00' * (padding - 4))
            
            zout.writestr(zinfo, data)

    out_buf.seek(0)
    with open(output_apk, 'wb') as fout:
        fout.write(out_buf.read())
    out_buf.close()

def sign_apk_python(unsigned_apk, signed_apk):
    # Signature cryptographique V1 par script stdlib avec openssl
    # (Copie le processus de bestmode)
    import base64, hashlib, tempfile
    
    # Éléments de signature
    KS_ALIAS_BYTES = KS_ALIAS.encode()
    
    # 1. Construction du Manifest
    file_digests = {}
    with zipfile.ZipFile(unsigned_apk, "r") as zin:
        file_list = [n for n in zin.namelist() if not n.startswith("META-INF/")]
        for name in file_list:
            data = zin.read(name)
            file_digests[name] = base64.b64encode(hashlib.sha256(data).digest()).decode()

    manifest_lines = ["Manifest-Version: 1.0\r\nCreated-By: React-Forge-Mobile\r\n\r\n"]
    for name, digest in sorted(file_digests.items()):
        manifest_lines.append(f"Name: {name}\r\nSHA-256-Digest: {digest}\r\n\r\n")
    manifest_content = "".join(manifest_lines).encode("utf-8")

    # 2. Construction Signature File (.SF)
    manifest_digest = base64.b64encode(hashlib.sha256(manifest_content).digest()).decode()
    sf_lines = [
        "Signature-Version: 1.0\r\n",
        "Created-By: React-Forge-Mobile\r\n",
        f"SHA-256-Digest-Manifest: {manifest_digest}\r\n\r\n"
    ]
    for name, digest in sorted(file_digests.items()):
        entry = f"Name: {name}\r\nSHA-256-Digest: {digest}\r\n\r\n"
        entry_digest = base64.b64encode(hashlib.sha256(entry.encode("utf-8")).digest()).decode()
        sf_lines.append(f"Name: {name}\r\nSHA-256-Digest: {entry_digest}\r\n\r\n")
    sf_content = "".join(sf_lines).encode("utf-8")

    # 3. Extraction Clé/Certificat via openssl
    tmp_key = tempfile.mktemp(suffix=".key.pem")
    tmp_cert = tempfile.mktemp(suffix=".cert.pem")
    tmp_certder = tempfile.mktemp(suffix=".cert.der")
    
    try:
        r1 = subprocess.run(["openssl", "pkcs12", "-in", KEYSTORE_PATH, "-nocerts", "-nodes", "-passin", f"pass:{KS_PASSWORD}", "-out", tmp_key], capture_output=True)
        r2 = subprocess.run(["openssl", "pkcs12", "-in", KEYSTORE_PATH, "-nokeys", "-clcerts", "-passin", f"pass:{KS_PASSWORD}", "-out", tmp_cert], capture_output=True)
        r3 = subprocess.run(["openssl", "x509", "-in", tmp_cert, "-outform", "DER", "-out", tmp_certder], capture_output=True)
        
        if r1.returncode != 0 or r2.returncode != 0 or r3.returncode != 0:
            raise Exception("Erreur lors de l'extraction des cles cryptographiques via openssl.")
            
        with open(tmp_key, "rb") as f: pkey_pem = f.read()
        with open(tmp_certder, "rb") as f: cert_der = f.read()
    finally:
        for tmp in [tmp_key, tmp_cert, tmp_certder]:
            if os.path.exists(tmp): os.remove(tmp)

    # 4. Signature avec clé privée
    tmp_sf = tempfile.mktemp(suffix=".sf")
    tmp_sig = tempfile.mktemp(suffix=".sig")
    try:
        with open(tmp_sf, "wb") as f: f.write(sf_content)
        tmp_key_f = tempfile.mktemp(suffix=".pem")
        with open(tmp_key_f, "wb") as f: f.write(pkey_pem)
        
        r_sign = subprocess.run(["openssl", "dgst", "-sha256", "-sign", tmp_key_f, "-out", tmp_sig, tmp_sf], capture_output=True)
        if r_sign.returncode != 0:
            raise Exception(f"Echec signature openssl : {r_sign.stderr.decode(errors='replace')}")
            
        with open(tmp_sig, "rb") as f: raw_sig = f.read()
    finally:
        for tmp in [tmp_sf, tmp_sig, tmp_key_f]:
            if os.path.exists(tmp): os.remove(tmp)

    # 5. Construction du bloc DER signature PKCS7 (CERT.RSA)
    def der_len(n):
        if n < 0x80: return bytes([n])
        nb = (n.bit_length() + 7) // 8
        return bytes([0x80 | nb]) + n.to_bytes(nb, 'big')
    def der_tlv(tag, content):
        return bytes([tag]) + der_len(len(content)) + content
    def der_seq(content): return der_tlv(0x30, content)
    def der_set(content): return der_tlv(0x31, content)
    def der_int(n): b = n.to_bytes((n.bit_length()+15)//8,'big'); return der_tlv(0x02, (b'\x00'+b if b[0]>=0x80 else b))
    def der_oid(oid_bytes): return der_tlv(0x06, oid_bytes)
    def der_octet(content): return der_tlv(0x04, content)

    OID_SHA256 = bytes.fromhex("608648016503040201")
    OID_RSA = bytes.fromhex("2a864886f70d010101")
    OID_PKCS7_DATA = bytes.fromhex("2a864886f70d010701")
    OID_SIGNED_DATA = bytes.fromhex("2a864886f70d010702")

    sha256_alg = der_seq(der_oid(OID_SHA256) + der_tlv(0x05, b''))
    rsa_alg = der_seq(der_oid(OID_RSA) + der_tlv(0x05, b''))

    signer_info = der_seq(
        der_int(1) +
        der_seq(cert_der[7:7+cert_der[7+1]+2] if len(cert_der)>9 else cert_der) +
        sha256_alg +
        rsa_alg +
        der_octet(raw_sig)
    )
    content_info = der_seq(der_oid(OID_PKCS7_DATA))
    signed_data = der_seq(
        der_int(1) +
        der_set(sha256_alg) +
        content_info +
        der_tlv(0xa0, cert_der) +
        der_set(signer_info)
    )
    pkcs7_der = der_seq(der_oid(OID_SIGNED_DATA) + der_tlv(0xa0, signed_data))

    # Assembler l'APK signé final
    out_buf = tempfile.SpooledTemporaryFile(max_size=100 * 1024 * 1024)
    with zipfile.ZipFile(unsigned_apk, "r") as zin:
        with zipfile.ZipFile(out_buf, "w", compression=zipfile.ZIP_STORED) as zout:
            for item in zin.infolist():
                if item.filename.startswith("META-INF/"):
                    continue
                zout.writestr(item, zin.read(item.filename))
            zout.writestr("META-INF/MANIFEST.MF", manifest_content)
            zout.writestr("META-INF/CERT.SF", sf_content)
            zout.writestr("META-INF/CERT.RSA", pkcs7_der)

    out_buf.seek(0)
    with open(signed_apk, "wb") as f:
        f.write(out_buf.read())
    out_buf.close()

def get_extension_packs_summary():
    import re
    ext_dir = os.path.join(BASE_DIR, "extensions")
    packs = []
    if not os.path.exists(ext_dir):
        return packs
    try:
        entries = os.listdir(ext_dir)
        for entry in entries:
            pack_dir = os.path.join(ext_dir, entry)
            if not os.path.isdir(pack_dir):
                continue
            if not entry.endswith("_pack") and entry != "interface_pack":
                continue
                
            manifest_path = os.path.join(pack_dir, "manifest.json")
            if not os.path.exists(manifest_path):
                continue
            try:
                with open(manifest_path, "r", encoding="utf-8") as f:
                    manifest = json.load(f)
                name = manifest.get("name", entry)
                description = manifest.get("description", "")
            except:
                continue
                
            inject_files = [f for f in os.listdir(pack_dir) if f.startswith("inject_") and f.endswith(".js")]
            if not inject_files:
                continue
                
            inject_path = os.path.join(pack_dir, inject_files[0])
            prd_count = 0
            try:
                with open(inject_path, "r", encoding="utf-8") as f:
                    content = f.read()
                matches = re.findall(r"(\w+):\s*`([\s\S]*?)`", content)
                for key, val in matches:
                    if key.startswith("prd_") or "CONTEXTE" in val or "MISSION" in val:
                        prd_count += 1
            except Exception as e:
                log(f"Error parsing inject file: {e}")
                continue
                
            if prd_count > 0:
                packs.append({
                    "id": entry,
                    "name": name,
                    "description": description,
                    "prdCount": prd_count
                })
    except Exception as e:
        log(f"Error getting extensions: {e}")
    return packs

def serve_spa(path):
    if path and path.startswith('api/'):
        return jsonify({"success": False, "error": "API non disponible en mode mobile autonome"}), 501

    out_dir = os.path.join(BASE_DIR, "out")
    if not path or path == "/":
        return send_from_directory(out_dir, 'index.html')
    
    # Tester si le fichier exact existe
    if os.path.exists(os.path.join(out_dir, path)):
        return send_from_directory(out_dir, path)
    
    # Ajouter .html si nécessaire (routage Next.js statique)
    if os.path.exists(os.path.join(out_dir, path + ".html")):
        return send_from_directory(out_dir, path + ".html")
        
    # Fallback pour le routage SPA Next.js
    return send_from_directory(out_dir, 'index.html')
